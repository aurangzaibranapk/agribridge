import Link from "next/link";
import { redirect } from "next/navigation";
import { createClient } from "@/lib/supabase/server";
import { MachineryForm } from "./machinery-form";

export default async function MachineryRequestPage() {
  const supabase = createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) redirect("/login");

  const { data: farmer } = await supabase.from("farmers").select("id").eq("user_id", user.id).single();
  if (!farmer) redirect("/login");

  return (
    <div className="mx-auto max-w-xl px-4 py-12">
      <Link href="/portal/dashboard" className="mb-4 inline-block text-sm text-surface-500 hover:text-brand-700">
        â† Back to Dashboard
      </Link>
      <h1 className="font-display text-2xl font-semibold text-surface-900">Machinery Rental Request</h1>
      <p className="mt-1 text-sm text-surface-500">Tell us which machine you need, and when and where you need it.</p>

      <div className="mt-6">
        <MachineryForm />
      </div>
    </div>
  );
}
